shadcn copies the components into your project rather than installing them as a dependency, so their classes are in your copy and nowhere else. They are written with your theme's tokens — `text-primary`, `text-muted-foreground`, `text-destructive` — so they match the rest of your UI without anything being set.

## Edit your copy

In `components/command-line.tsx`, each kind of word takes its classes from the `tone` map at the top of the file:

```tsx
const tone = {
  program: 'font-medium text-foreground',
  subcommand: 'text-foreground',
  flag: 'text-primary',
  value: 'text-sky-700 dark:text-sky-300',
  operator: 'text-muted-foreground',
  redirect: 'text-muted-foreground',
  word: 'text-foreground/80',
  unread: 'text-destructive underline decoration-wavy',
}
```

<OpenInV0 item="command-line" />

A flag that widened the answer adds `rounded-sm bg-primary/15`, and the program another one hands over to adds a dotted underline; both are in `commandLinePartClassName`, just below the map.

In `components/sentence.tsx`, each value is drawn by the `Value` element — shadcn's `Badge`, set to sit inline and wrap with the sentence, around a `<code>`:

```tsx
<Badge
  asChild
  variant="outline"
  className="inline rounded-sm border-transparent bg-primary/10 px-1 py-0.5 font-mono text-[0.85em] font-normal whitespace-normal text-primary"
>
```

<OpenInV0 item="sentence" />

Change these and every use of the component follows.

## Override one use

For one place that needs something else, pass `renderValue` to [`Sentence`](/docs/react/sentence) or `renderPart` to [`CommandLine`](/docs/react/command-line), and draw with whatever classes you like. Here each value is drawn by its `role`: coloured in the command, underlined with an icon in the sentence, and pointing at either lights the other:

<ComponentPreview name="sentence-by-role-demo">

```tsx
'use client'

import { useMemo, useState } from 'react'
import { linePartsOf, type Explanation, type Role } from 'cmd-facts'
import { Cpu, FileText, Globe, MousePointer2, RotateCcw, Server, User } from 'lucide-react'
import type { LucideIcon } from 'lucide-react'
import { CommandLine } from '@/components/command-line'
import { Sentence } from '@/components/sentence'
import { Button } from '@/components/ui/button'
import { Card, CardContent, CardDescription, CardFooter, CardHeader, CardTitle } from '@/components/ui/card'
import { useExplain } from '@/hooks/use-explain'

const command = 'curl -o dump.sql https://db.example.com/export && chown alice dump.sql && ssh admin@db.internal'

// One look per kind of value, so a reader tells a host from a file at a glance: a colour for the word in the
// command, an underline and an icon for the value in the sentence, and a wash for both when either is pointed at.
// A role not listed keeps the default.
type Look = { icon: LucideIcon; text: string; underline: string; wash: string }
const byRole: Partial<Record<Role, Look>> = {
  url: { icon: Globe, text: 'text-sky-700 dark:text-sky-300', underline: 'decoration-sky-500', wash: 'bg-sky-400/15' },
  host: { icon: Server, text: 'text-violet-700 dark:text-violet-300', underline: 'decoration-violet-500', wash: 'bg-violet-400/15' },
  path: { icon: FileText, text: 'text-amber-700 dark:text-amber-300', underline: 'decoration-amber-500', wash: 'bg-amber-400/15' },
  user: { icon: User, text: 'text-emerald-700 dark:text-emerald-300', underline: 'decoration-emerald-500', wash: 'bg-emerald-400/15' },
  pid: { icon: Cpu, text: 'text-rose-700 dark:text-rose-300', underline: 'decoration-rose-500', wash: 'bg-rose-400/15' },
}

/**
 * Which words of the line each value in the sentence was read from, keyed by the value's place in `parts`. The
 * sentence can name values in another order than the line, so each one takes the first unclaimed word of its role
 * spelled the way it was typed.
 */
function wordsOf(explanation: Explanation) {
  const line = linePartsOf(command, explanation)
  const claimed = new Map<number, number>()
  const words = new Map<number, number[]>()
  explanation.parts.forEach((part, index) => {
    if (part.kind !== 'values') return
    const typed = part.written ?? part.values
    words.set(
      index,
      typed.flatMap((written) => {
        const at = line.findIndex(
          (word, i) =>
            !claimed.has(i) &&
            word.kind === 'value' &&
            word.role === part.role &&
            word.text.replace(/^['"]|['"]$/g, '') === written,
        )
        if (at < 0) return []
        claimed.set(at, index)
        return [at]
      }),
    )
  })
  // and back: which value each word of the line became
  return { words, valueOf: claimed }
}

export default function Page() {
  const explanation = useExplain(command)
  const { words, valueOf } = useMemo(() => wordsOf(explanation), [explanation])
  // the value being pointed at, from the sentence or from the line
  const [pointed, setPointed] = useState<number | null>(null)
  const [decision, setDecision] = useState<'allowed' | 'denied' | null>(null)
  const roleOf = (index: number) => {
    const part = explanation.parts[index]
    return part?.kind === 'values' ? part.role : 'unknown'
  }

  return (
    <main className="mx-auto max-w-xl p-8">
      <Card>
        <CardHeader>
          <CardTitle>Allow this command?</CardTitle>
          <CardDescription>An agent wants to run:</CardDescription>
        </CardHeader>
        <CardContent className="flex flex-col gap-4">
          <CommandLine
            command={command}
            of={explanation}
            className="rounded-md bg-muted px-3 py-2 text-sm"
            renderPart={(part, i) => {
              if (part.kind === 'space') return part.text
              const value = valueOf.get(i)
              if (value === undefined) return <span className="text-muted-foreground">{part.text}</span>
              const look = byRole[roleOf(value)]
              const lit = pointed === value
              return (
                <span
                  onMouseEnter={() => setPointed(value)}
                  onMouseLeave={() => setPointed(null)}
                  className={`cursor-help rounded-sm transition-colors ${look?.text ?? ''} ${lit ? (look?.wash ?? 'bg-accent') : ''}`}
                >
                  {part.text}
                </span>
              )
            }}
          />
          <p className="leading-relaxed">
            <Sentence
              of={explanation}
              renderValue={(v) => {
                const look = byRole[v.role]
                const point = () => setPointed(v.index)
                const leave = () => setPointed(null)
                const lit = pointed === v.index && (words.get(v.index)?.length ?? 0) > 0
                return (
                  <span
                    tabIndex={0}
                    title={`${v.role}: ${v.written.join(', ')}`}
                    onMouseEnter={point}
                    onMouseLeave={leave}
                    onFocus={point}
                    onBlur={leave}
                    className={`cursor-help rounded-sm pl-1 font-mono text-[0.9em] whitespace-nowrap underline decoration-dotted decoration-2 underline-offset-4 outline-none transition-colors focus-visible:ring-1 focus-visible:ring-current ${
                      look?.underline ?? ''
                    } ${lit ? (look?.wash ?? 'bg-accent') : ''}`}
                  >
                    {look && <look.icon aria-hidden className={`mr-0.5 inline size-[0.9em] align-[-0.1em] ${look.text}`} />}
                    {v.text}
                  </span>
                )
              }}
            />
          </p>
          <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
            <MousePointer2 aria-hidden className="size-3.5" />
            Point at an underlined value or a coloured word to see its pair.
          </p>
        </CardContent>
        <CardFooter className="justify-end gap-2">
          {decision ? (
            <>
              <p className="text-sm text-muted-foreground">{decision === 'allowed' ? 'Allowed.' : 'Denied.'}</p>
              <Button variant="outline" size="sm" onClick={() => setDecision(null)}>
                <RotateCcw aria-hidden />
                Reset
              </Button>
            </>
          ) : (
            <>
              <Button variant="outline" onClick={() => setDecision('denied')}>
                Deny
              </Button>
              <Button onClick={() => setDecision('allowed')}>Allow</Button>
            </>
          )}
        </CardFooter>
      </Card>
    </main>
  )
}
```

</ComponentPreview>

`className` on either component goes to its outer element, merged with `cn`.

## Plain CSS

The default pieces carry what they are as attributes — `data-kind` and `data-role` on the line, `data-role` on a sentence's values — so a stylesheet can reach them without touching the components:

```css
[data-slot='command-line'] [data-kind='flag'] { color: hotpink; }
[data-slot='sentence'] [data-role='url'] { color: skyblue; }
```
